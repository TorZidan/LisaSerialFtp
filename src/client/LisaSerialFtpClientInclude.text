{ LISA-FTP -- fragment of LisaSerialFtpClient.text, spliced in by the      }
{ Lisa Pascal file-inclusion directive (the "I" directive) naming this     }
{ file.                                                                    }
{                                                                          }
{ Holds the Workshop client's code for the local-file commands '!list'    }
{ (local volume listing), '!delete' and '!volume' (the Office System       }
{ client has its own, in LosSerialFtpClient.text); EntryKind, Pad2 and     }
{ DateStr are in LisaSerialFtpUtil.text, shared by both:                   }
{   DoListVol  -- the '!list' handler itself                               }
{   AskYes     -- a Y/N question (also used by 'delete', later in          }
{                 LisaSerialFtpClient.text)                                }
{   DoDelLocal -- the '!delete' handler                                    }
{   DoVolCmd   -- the '!volume' handler (list the mounted volumes, or      }
{                 make one the working volume)                             }
{                                                                          }
{ This is NOT a standalone program: no PROGRAM statement, no USES, no      }
{ final END.  It is included at the point of the I-inclusion directive     }
{ in LisaSerialFtpClient.text -- after all program-scope declarations,     }
{ so string255 is in scope -- and uses the syscall unit's                  }
{ GET_WORKING_DIR / SET_WORKING_DIR / LOOKUP / RESET_CATALOG /             }
{ LOOKUP_NEXT_ENTRY / KILL_OBJECT, and the program's kbd, ecode,           }
{ IntToStr, StrToInt and UpperStr.  Works                                  }
{ purely locally: no serial link or server needed.                         }
{****************************************************************************}

{ List the catalog of the default volume -- every file on it, not just
  one directory (the same scan UltraLOS's ListFile does).  Without an
  argument, all entries on the volume are listed; with one, only
  entries whose name begins with the argument (the catalog prefix match
  is case-insensitive).  Paged by PageStart / PageMore (LisaSerialFtpClient.text):
  'Type SPACE to continue, CLEAR to exit.' after every 30 lines.  One
  line per entry, with the attributes LOOKUP_NEXT_ENTRY's Q_Info carries
  (no second LOOKUP per entry).  Works purely locally -- no serial
  link or server needed.

  The layout is like the Workshop File Manager's (List), with room for
  the longest Lisa name (MAXNAME, 32 characters), so no name is cut:
    Contents of -vol-=
    Filename                            Size Psize   Last-Mod-Date  Creation-Date Attr
    --------                            ---- -----   -------------  ------------- ----
    ALERT                              18432    36  06/22/84-14:08 06/22/84-14:06
  Attr shows the entry type when it is not a plain file (catalog, link,
  ...) and 'protected' for a protected file.  A line is 77 characters
  (87 with 'protected'): within the 88 columns of the Lisa console, so
  it never wraps.

  LOOKUP_NEXT_ENTRY does the prefix match (case-insensitive).  A
  trailing '=' (the Workshop's wildcard, as in 'ALEX=') is accepted and
  dropped first.  The volume name in the heading is the first part of
  the working directory's path ('-#12' -> '-#12-'). }
PROCEDURE DoListVol (arg: string255);
VAR
  err    : integer;
  path   : pathname;
  vol    : pathname;
  prefix : e_name;
  uprefix: string255;          { the argument, cleaned up into prefix }
  q      : Q_Info;
  outl   : string255;
  tn     : string255;
  attr   : string255;
  dm, dcr: string255;          { last-modified and creation dates }
  j      : integer;
  n      : longint;
  more   : boolean;
BEGIN
  err := 0;
  path := '';
  GET_WORKING_DIR(err, path);
  IF err <> 0 THEN
  BEGIN
    WRITELN('!list: cannot get working directory: ', err);
    EXIT(DoListVol);
  END;
  { The volume is the first component of the working directory's path:
    '-#12' or '-#12-SUBDIR' -> '-#12-'.  An empty path gives '-'. }
  j := 2;
  WHILE (j <= LENGTH(path)) AND (path[j] <> '-') DO j := j + 1;
  IF LENGTH(path) = 0 THEN
    vol := '-'
  ELSE
    vol := concat(COPY(path, 1, j - 1), '-');
  { the prefix: without a trailing '=' (Workshop wildcard), at most the
    32 characters of an entry name }
  uprefix := arg;
  IF LENGTH(uprefix) > 0 THEN
    IF uprefix[LENGTH(uprefix)] = '=' THEN
      uprefix := COPY(uprefix, 1, LENGTH(uprefix) - 1);
  IF LENGTH(uprefix) > 32 THEN
    uprefix := COPY(uprefix, 1, 32);
  prefix := uprefix;
  err := 0;
  RESET_CATALOG(err, path);
  IF err <> 0 THEN
  BEGIN
    WRITELN('!list: cannot reset catalog of ', path, ': ', err);
    EXIT(DoListVol);
  END;
  PageStart;
  more := PageMore;
  WRITELN('Contents of ', vol, prefix, '=');
  IF PageMore THEN
    WRITELN('Filename                            Size Psize   Last-Mod-Date  Creation-Date Attr')
  ELSE
    EXIT(DoListVol);
  IF PageMore THEN
    WRITELN('--------                            ---- -----   -------------  ------------- ----')
  ELSE
    EXIT(DoListVol);
  n := 0;
  REPEAT
    err := 0;
    LOOKUP_NEXT_ENTRY(err, prefix, q);
    IF err = 0 THEN
    BEGIN
      n := n + 1;
      outl := q.name;                     { at most MAXNAME (32) characters }
      FOR j := LENGTH(outl) + 1 TO MaxEName DO
        outl := concat(outl, ' ');
      attr := '';
      IF q.etype <> fileentry THEN
      BEGIN
        EntryKind(q.etype, tn);
        attr := tn;
      END;
      IF q.protected THEN
        IF attr = '' THEN
          attr := 'protected'
        ELSE
          attr := concat(attr, ' protected');
      IF attr <> '' THEN
        attr := concat(' ', attr);        { after the dates, if any }
      DateStr(q.DTM, dm);
      DateStr(q.DTC, dcr);
      more := PageMore;
      IF more THEN
        { Psize: 512-byte blocks.  Q_Info.psize is the physical size in
          BYTES (18432 for 36 blocks); rounded up, in case it is not a
          whole number of blocks. }
        WRITELN(outl, q.size:8, (q.psize + 511) DIV 512:6, '  ', dm, ' ', dcr, attr);
    END;
  UNTIL (err <> 0) OR NOT more;
  IF NOT more THEN EXIT(DoListVol);       { CLEAR at a page prompt }
  IF err <> 848 THEN
    IF PageMore THEN
      WRITELN('!list: catalog error: ', err);
  IF PageMore THEN
    WRITELN(n, ' entries');
END;

{ Ask question Q and wait for Y or N (read from kbd, no echo; other keys
  are ignored).  TRUE for Y.  In a Workshop exec file the answer comes
  from the file (the test macro answers 'Y' on the next line). }
FUNCTION AskYes (Q: string255): BOOLEAN;
VAR ch: CHAR;
BEGIN
  WRITE(Q, ' (Y/N) ');
  REPEAT
    READ(kbd, ch);
  UNTIL (ch = 'Y') OR (ch = 'y') OR (ch = 'N') OR (ch = 'n');
  WRITELN(ch);
  AskYes := (ch = 'Y') OR (ch = 'y');
END;

{ '!delete <local>' / '!del': delete a file on the Lisa (KILL_OBJECT),
  after a Y/N question.  Whether the file exists is not checked first:
  KILL_OBJECT's error says so. }
PROCEDURE DoDelLocal (name: string255);
VAR p: pathname;
BEGIN
  IF name = '' THEN
  BEGIN
    WRITELN('usage: !delete <local>   (or !del)');
    EXIT(DoDelLocal);
  END;
  IF NOT AskYes(concat('delete ', name, ' on the Lisa?')) THEN
  BEGIN
    WRITELN('not deleted');
    EXIT(DoDelLocal);
  END;
  p := name;
  ecode := 0;
  KILL_OBJECT(ecode, p);
  IF ecode > 0 THEN
    WRITELN('cannot delete ', name, ': error ', ecode)
  ELSE
    WRITELN('deleted ', name);
END;

{ '!volume [<n> | -vol-]' / '!vol': without an argument, list the
  mounted Lisa volumes (the working volume first, then the usual device
  names), numbered, with their free blocks; with a number from that list
  or a volume name, make that volume the working volume (SET_WORKING_DIR).
  Local file names without a volume ('!list', '!delete', get, put) are
  then on that volume.  The working directory belongs to this program:
  the Workshop's own working volume is unchanged after quit.  Volumes
  with the same free count are marked, as they may be one disk under two
  names.  The list is made again for each '!volume' command, so a number
  always means the same volume as in the last listing (unless a disk was
  mounted or removed in between). }
PROCEDURE DoVolCmd (arg: string255);
CONST
  MaxVols = 10;                { at most this many volumes are listed }
VAR
  volList: ARRAY [1..MaxVols] OF e_name;    { '-#12-', '-UPPER-', ... }
  volFree: ARRAY [1..MaxVols] OF LONGINT;
  volCnt : integer;
  workVol: e_name;             { the working volume, as '-name-' }
  path   : pathname;
  err, j : integer;
  k      : LONGINT;
  allDig : boolean;
  choice : string255;

  { add the volume pfx ('-name-') to the list if it is mounted and not
    listed yet; print its line when show is true }
  PROCEDURE ProbeVol (pfx: e_name; show: boolean);
  VAR fsi: fs_info; ec, dup, m: integer; p: pathname; ns: string255;
  BEGIN
    IF volCnt >= MaxVols THEN EXIT(ProbeVol);
    FOR m := 1 TO volCnt DO
      IF volList[m] = pfx THEN EXIT(ProbeVol);
    p := pfx;
    ec := 0;
    LOOKUP(ec, p, fsi);
    IF ec <> 0 THEN EXIT(ProbeVol);
    IF NOT fsi.mounted THEN EXIT(ProbeVol);
    dup := 0;
    FOR m := 1 TO volCnt DO
      IF volFree[m] = fsi.freecount THEN dup := m;
    volCnt := volCnt + 1;
    volList[volCnt] := pfx;
    volFree[volCnt] := fsi.freecount;
    IF show THEN
    BEGIN
      WRITE('  ', volCnt:2, '  ', pfx, '  ', fsi.freecount, ' free blocks (~',
            fsi.freecount DIV 2, ' KB)');
      IF pfx = workVol THEN WRITE('  [working volume]');
      IF dup > 0 THEN
      BEGIN
        IntToStr(dup, ns);
        WRITE('  (same free count as #', ns, ')');
      END;
      WRITELN;
    END;
  END;

  PROCEDURE ScanVols (show: boolean);
  BEGIN
    IF show THEN WRITELN('Mounted Lisa volumes:');
    volCnt := 0;
    ProbeVol(workVol, show);
    ProbeVol('-#12-', show);
    ProbeVol('-#2#1-', show);
    ProbeVol('-#2#2-', show);
    ProbeVol('-#3#1-', show);
    ProbeVol('-#3#2-', show);
    ProbeVol('-#1#1-', show);
    ProbeVol('-#1#2-', show);
    ProbeVol('-LOWER-', show);
    ProbeVol('-UPPER-', show);
  END;

BEGIN
  { the working volume: the first component of the working directory's
    path, '-#12' or '-#12-SUBDIR' -> '-#12-' (as in DoListVol) }
  err := 0;
  path := '';
  GET_WORKING_DIR(err, path);
  IF (err <> 0) OR (LENGTH(path) = 0) THEN
    workVol := '-'
  ELSE
  BEGIN
    j := 2;
    WHILE (j <= LENGTH(path)) AND (path[j] <> '-') DO j := j + 1;
    workVol := concat(COPY(path, 1, j - 1), '-');
  END;
  IF arg = '' THEN
  BEGIN
    ScanVols(TRUE);
    WRITELN('Current working volume: ', workVol,
            '  (change with: !volume <number | -vol->)');
    EXIT(DoVolCmd);
  END;
  { a number from the list }
  allDig := TRUE;
  FOR j := 1 TO LENGTH(arg) DO
    IF (arg[j] < '0') OR (arg[j] > '9') THEN allDig := FALSE;
  IF allDig THEN
  BEGIN
    ScanVols(FALSE);
    k := StrToInt(arg);
    IF (k < 1) OR (k > volCnt) THEN
    BEGIN
      WRITELN('not one of the listed volume numbers (see !volume)');
      EXIT(DoVolCmd);
    END;
    choice := volList[k];
  END
  ELSE
  BEGIN
    { a volume name: 'UPPER', '-UPPER' or '-UPPER-' }
    choice := arg;
    IF choice[1] <> '-' THEN choice := concat('-', choice);
    IF choice[LENGTH(choice)] <> '-' THEN choice := concat(choice, '-');
    UpperStr(choice);
    IF LENGTH(choice) > max_ename THEN
    BEGIN
      WRITELN('volume name too long');
      EXIT(DoVolCmd);
    END;
  END;
  { the working directory is the volume's path without the last '-' }
  path := COPY(choice, 1, LENGTH(choice) - 1);
  err := 0;
  SET_WORKING_DIR(err, path);
  IF err <> 0 THEN
    WRITELN('cannot make ', choice, ' the working volume: error ', err)
  ELSE
    WRITELN('Working volume is now ', choice);
END;
