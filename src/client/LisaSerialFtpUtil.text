function ByteVal(b: ByteT): integer;
begin
   if b < 0 then ByteVal := b + 256
   else ByteVal := b;
end;

function ToByte(v: longint): integer;
begin
   if v > 127 then ToByte := v - 256
   else ToByte := v;
end;

procedure MoveBytes(SRC, DST: BytePtr; N: integer);
begin
   if N > 0 then MOVEB(SRC, DST, N);
end;

procedure IntToStr(N: longint; var S: Str255);
var m: longint; d: integer; t: Str255;
begin
   if N < 0 then begin
      m := -N;
      S := '-';
   end
   else begin
      m := N;
      S := '';
   end;
   if m = 0 then
      S := concat(S, '0')
   else begin
      t := '';
      while m > 0 do begin
         d := m mod 10;
         t := concat(copy('0123456789', d + 1, 1), t);
         m := m div 10;
      end;
      S := concat(S, t);
   end;
end;

function StrToInt(S: Str255): longint;
var n: longint; j: integer;
begin
   n := 0;
   for j := 1 to length(S) do
      if (S[j] >= '0') and (S[j] <= '9') then
         n := n * 10 + ord(S[j]) - ord('0');
   StrToInt := n;
end;

procedure IntToHex(N: integer; var S: Str255);
const hexd = '0123456789ABCDEF';
var m, dv: longint; j, d: integer;
begin
   m := N;
   if m < 0 then m := m + 65536;
   S := '';
   dv := 4096;
   for j := 3 downto 0 do begin
      d := (m div dv) mod 16;
      S := concat(S, copy(hexd, d + 1, 1));
      dv := dv div 16;
   end;
end;

procedure UpperStr(var S: Str255);
var j: integer;
begin
   for j := 1 to length(S) do
      if (S[j] >= 'a') and (S[j] <= 'z') then
         S[j] := chr(ord(S[j]) - 32);
end;

procedure TrimStr(var S: Str255);
begin
   while length(S) > 0 do
      if S[1] = ' ' then delete(S, 1, 1)
      else if S[length(S)] = ' ' then delete(S, length(S), 1)
      else exit(TrimStr);
end;

procedure AddChar(var S: Str255; C: char);
var n: integer;
begin
   n := length(S);
   if n < MaxLine then begin
      S[n + 1] := C;
      S[0] := chr(n + 1);
   end;
end;

procedure PadRight(var S: Str255; width: integer);
begin
   while (length(S) < width) and (length(S) < MaxLine) do
      AddChar(S, ' ');
end;

procedure PadLeftNum(N: longint; width: integer; var S: Str255);
var t: Str255;
begin
   IntToStr(N, t);
   S := '';
   while (length(S) + length(t) < width) and (length(S) < MaxLine) do
      AddChar(S, ' ');
   S := concat(S, t);
end;

procedure EntryKind(t: entrytype; var s: Str255);
begin
   s := 'other';
   if t = catentry then s := 'catalog'
   else if t = linkentry then s := 'link'
   else if t = fileentry then s := 'file'
   else if t = pipeentry then s := 'pipe'
   else if t = ecentry then s := 'ec'
   else if t = killedentry then s := 'killed';
end;

procedure Pad2(N: longint; var S: Str255);
var t: Str255;
begin
   IntToStr(N, t);
   if N < 10 then t := concat('0', t);
   S := concat(S, t);
end;

procedure DateStr(T: longint; var S: Str255);
var hi, lo, days, secs, cyc, dc, y, dd, year, m, ml: longint;
begin
   if T < 0 then begin
      T := (T + 1073741824) + 1073741824;
      hi := T div 128 + 16777216;
   end
   else
      hi := T div 128;
   lo := T mod 128;
   days := hi div 675;
   secs := (hi mod 675) * 128 + lo;
   cyc := days div 1461;
   dc := days mod 1461;
   y := dc div 365;
   dd := dc mod 365;
   if y = 4 then begin
      y := 3;
      dd := 365;
   end;
   year := 1901 + cyc * 4 + y;
   m := 1;
   ml := 31;
   while dd >= ml do begin
      dd := dd - ml;
      m := m + 1;
      if m = 2 then begin
         if year mod 4 = 0 then ml := 29 else ml := 28;
      end
      else if (m = 4) or (m = 6) or (m = 9) or (m = 11) then
         ml := 30
      else
         ml := 31;
   end;
   S := '';
   Pad2(m, S);
   S := concat(S, '/');
   Pad2(dd + 1, S);
   S := concat(S, '/');
   Pad2(year mod 100, S);
   S := concat(S, '-');
   Pad2(secs div 3600, S);
   S := concat(S, ':');
   Pad2((secs mod 3600) div 60, S);
end;
