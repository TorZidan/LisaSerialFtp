{==============================================================================}
{                                                                              }
{  LosSerialFtpClient.text -- Serial port file transfer desktop client         }
{  application for the vintage Apple Lisa computer. Runs in the Apple Lisa     } 
{  Office System (LOS 3.1) desktop environment.                                }
{  Transfers text and binary files, lists directories, and deletes files       }
{  between an Apple Lisa (Serial Port B, RS232B) and a modern computer         }
{  running LisaSerialFtpServer.py or the web-based server.                     }
{                                                                              }
{  FEATURE PARITY with the LisaSerialFtpClient.text which runs in the Workshop }
{  environment.                                                                }
{                                                                              }
{  SHARED CODE: the serial port routines, the CRC, the chunk protocol, the     }
{  transfer engine and the server commands are shared with the Lisa Workshop   }
{  client (LisaSerialFtpClient.text), in LisaSerialFtpDecl.text,               }
{  LisaSerialFtpUtil.text, LisaSerialFtpSerial.text, LisaSerialFtpProto.text,  }
{  LisaSerialFtpXfer.text and LisaSerialFtpCmds.text, included with the I      }
{  directive (keep them on the same volume).  This file holds the Office       }
{  System side, and what the shared code calls (ConWr, SetStatus, ...).        }
{                                                                              }
{  THE OFFICE SYSTEM & QUICKDRAW SIDE                                          }
{    - Filer handshake (InitGraf, OpenWM, Initialize, TakeWindow, TellFiler,   }
{      SYS_TERMINATE -> ImDying);                                              }
{    - Pull-down menus + alerts loaded from <tool>PHRASE via InitMP;           }
{    - Window Manager dialog band (DialogFolder / DialogHeight) for menu       }
{      prompts, Y/N confirmations, and listing pagination;                     }
{    - QuickDraw window with:                                                  }
{        * top info header bar with rounded CONNECTED / OFFLINE status pill,   }
{          baud rate, working volume, and timeout multiplier;                  }
{        * scrollable monospaced console pane (240 lines in a private data     }
{          segment) with interactive '> ' command prompt, blinking XOR caret,  }
{          command history (Up/Down), vertical scroll bar (WMLsb), and         }
{          resizable grow box (WMLgrow);                                       }
{        * bottom status bar with live QuickDraw graphical progress bar        }
{          during file transfers;                                              }
{    - Event-queue polling during waits and transfers so Apple-period aborts   }
{      cleanly while window updates and Filer events stay alive.               }
{                                                                              }
{  THE PROTOCOL SIDE                                                           }
{    - Speaks LISA-SERIAL-FTP/v1.0 over RS232B (up to 57600 baud, 8N1);        }
{    - Stop-and-wait 1016-byte CRC-16 chunks, automatic NAK/retransmit loss    }
{      recovery, held ACKs around disk writes, atomic temp-file rename on      }
{      verified GET, 68000 assembly CRC16A / SUM16A / MOVEB / DECODEP.         }
{                                                                              }
{                                                                              }
{                                                                              }
{==============================================================================}

{ Resolve a user-typed local filename against curVol when bare }
procedure FullPath(nm: Str255; var outPath: pathname);
var s: Str255;
begin
   s := nm;
   TrimStr(s);
   if length(s) = 0 then
      outPath := ''
   else if s[1] = '-' then
      outPath := s
   else if s[1] = '#' then
      outPath := concat('-', s)
   else
      outPath := concat(curVol, s);
end;

procedure InstallPort(folder: WindowPtr);
begin
   if folder <> nil then begin
      theFolder := folder;
      SetPort(theFolder);
   end;
end;

function MaxTopLn: integer;
var m: integer;
begin
   m := conCnt - visRows + 1;
   if m < 1 then m := 1;
   MaxTopLn := m;
end;

function CalcElev(lineNo: integer): integer;
var travel: integer;
begin
   travel := MaxTopLn - 1;
   if (lineNo <= 1) or (travel <= 0) then CalcElev := 0
   else if lineNo - 1 >= travel then CalcElev := 1000
   else CalcElev := (ord4(1000) * (lineNo - 1)) div travel;
end;

function CalcTopFromElev(elev: integer): integer;
var travel: integer;
begin
   travel := MaxTopLn - 1;
   if travel <= 0 then CalcTopFromElev := 1
   else CalcTopFromElev := 1 + (ord4(travel) * elev) div 1000;
end;

procedure UseTermFont;
var fId: TLFntId; fInfo: FontInfo; errNum: integer; fam: integer;
begin
   if fontChoice = miFont20 then fam := p20Tile
   else if fontChoice = miFont12 then fam := p12Tile
   else fam := p15Tile;

   fId.Fam := fam;
   fId.SetEFace := [];
   fId.Dev := devScreen;
   if not FMFontMetrics(fId, fInfo, errNum) then begin
      fam := p12Tile;
      fId.Fam := fam;
      if not FMFontMetrics(fId, fInfo, errNum) then fam := sysText;
   end;
   termFam := fam;
   if myWindow <> nil then begin
      SetPort(myWindow);
      TextFont(termFam);
      TextFace([]);
      GetFontInfo(fInfo);
      lineHt := fInfo.ascent + fInfo.descent + fInfo.leading;
      if lineHt < 8 then lineHt := 12;
      ascentPx := fInfo.ascent;
      charW := CharWidth('0');
      if charW < 4 then charW := 7;
   end;
end;

procedure RecompGeom;
var body: Rect;
begin
   if myWindow = nil then exit(RecompGeom);
   InstallPort(myWindow);
   UseTermFont;

   hdrRect := myWindow^.portRect;
   hdrRect.bottom := hdrRect.top + HdrBarH;

   body := myWindow^.portRect;
   body.top := hdrRect.bottom;
   body.right := body.right - (DhSbox - 1);
   body.bottom := body.bottom - (DvSbox - 1);
   if body.bottom < body.top + lineHt then
      body.bottom := body.top + lineHt;
   if body.right < body.left + 40 then
      body.right := body.left + 40;

   termRect := body;
   InsetRect(termRect, PadH, PadV);
   visRows := (termRect.bottom - termRect.top) div lineHt;
   if visRows < 1 then visRows := 1;
   termRect.bottom := termRect.top + visRows * lineHt;

   if topLine > MaxTopLn then topLine := MaxTopLn;
   if topLine < 1 then topLine := 1;
end;

procedure ClipTerm;
begin
   if myWindow <> nil then begin
      SetPort(myWindow);
      ClipRect(termRect);
   end;
end;

procedure CaretPos(var pt: Point);
var s: ConLine; row: integer;
begin
   row := conCnt - topLine;
   pt.v := termRect.top + row * lineHt + ascentPx;
   s := conBuf^[conCnt];
   TextFont(termFam);
   TextFace([]);
   pt.h := termRect.left + StringWidth(s);
   if pt.h > termRect.right - 2 then pt.h := termRect.right - 2;
end;

procedure CaretSet(wantVis: boolean);
var pt: Point;
begin
   if (not isActive) or (myWindow = nil) or (not atPrompt) then begin
      caretVis := false;
      exit(CaretSet);
   end;
   if (conCnt < topLine) or (conCnt >= topLine + visRows) then begin
      caretVis := false;
      exit(CaretSet);
   end;
   if caretVis = wantVis then exit(CaretSet);
   InstallPort(myWindow);
   ClipTerm;
   CaretPos(pt);
   PenNormal;
   PenMode(patXor);
   MoveTo(pt.h, pt.v - ascentPx + 1);
   Line(0, lineHt - 2);
   PenNormal;
   caretVis := wantVis;
end;

procedure SyncElev;
var newElev: integer;
begin
   newElev := CalcElev(topLine);
   if newElev <> elevPos then begin
      elevPos := newElev;
      if isActive and sbMade and (myWindow <> nil) then begin
         InstallPort(myWindow);
         SetupMvThumb(rightSBox);
         ClipRect(myWindow^.portRect);
         MoveThumb(elevPos);
         PenRestore;
         ClipTerm;
      end;
   end;
end;

procedure PaintRow(lnIdx: integer);
var row, y: integer; band: Rect; s: ConLine;
begin
   if myWindow = nil then exit(PaintRow);
   if (lnIdx < topLine) or (lnIdx >= topLine + visRows) then exit(PaintRow);
   if lnIdx = conCnt then CaretSet(false);
   InstallPort(myWindow);
   ClipTerm;
   TextFont(termFam);
   TextFace([]);
   row := lnIdx - topLine;
   band := termRect;
   band.top := termRect.top + row * lineHt;
   band.bottom := band.top + lineHt;
   EraseRect(band);
   if (lnIdx >= 1) and (lnIdx <= conCnt) then begin
      s := conBuf^[lnIdx];
      if length(s) > 0 then begin
         y := band.top + ascentPx;
         MoveTo(termRect.left, y);
         DrawString(s);
      end;
   end;
end;

procedure PaintTerm;
var r, idx: integer; clrRect: Rect;
begin
   if myWindow = nil then exit(PaintTerm);
   CaretSet(false);
   InstallPort(myWindow);
   clrRect := myWindow^.portRect;
   clrRect.top := hdrRect.bottom;
   clrRect.right := clrRect.right - (DhSbox - 1);
   clrRect.bottom := clrRect.bottom - (DvSbox - 1);
   ClipRect(clrRect);
   EraseRect(clrRect);
   ClipTerm;
   TextFont(termFam);
   TextFace([]);
   for r := 0 to visRows - 1 do begin
      idx := topLine + r;
      if (idx >= 1) and (idx <= conCnt) then
         if length(conBuf^[idx]) > 0 then begin
            MoveTo(termRect.left, termRect.top + r * lineHt + ascentPx);
            DrawString(conBuf^[idx]);
         end;
   end;
end;

procedure ScrollToLn(newTop: integer);
var delta, k: integer;
begin
   if newTop > MaxTopLn then newTop := MaxTopLn;
   if newTop < 1 then newTop := 1;
   delta := newTop - topLine;
   if delta = 0 then begin
      SyncElev;
      exit(ScrollToLn);
   end;
   CaretSet(false);
   if (myWindow <> nil) and (abs(delta) < visRows) then begin
      InstallPort(myWindow);
      ClipTerm;
      ScrollRect(termRect, 0, -delta * lineHt, scrollRgn);
      topLine := newTop;
      if delta > 0 then begin
         for k := topLine + visRows - delta to topLine + visRows - 1 do
            PaintRow(k);
      end
      else begin
         for k := topLine to topLine + (-delta) - 1 do
            PaintRow(k);
      end;
   end
   else begin
      topLine := newTop;
      PaintTerm;
   end;
   SyncElev;
end;

procedure SnapBottom;
begin
   if topLine <> MaxTopLn then
      ScrollToLn(MaxTopLn);
end;

{ Append text to the current console line (wrapping at ConCols) }
procedure ConWr(s: Str255);
var i, room, take: integer; cur: ConLine;
begin
   if length(s) = 0 then exit(ConWr);
   CaretSet(false);
   i := 1;
   while i <= length(s) do begin
      cur := conBuf^[conCnt];
      room := ConCols - length(cur);
      if room <= 0 then begin
         { wrap onto a fresh line }
         if conCnt < conCap then
            conCnt := conCnt + 1
         else begin
            for take := 1 to conCap - 1 do
               conBuf^[take] := conBuf^[take + 1];
         end;
         conBuf^[conCnt] := '';
         cur := '';
         room := ConCols;
         if myWindow <> nil then ScrollToLn(MaxTopLn);
      end;
      take := length(s) - i + 1;
      if take > room then take := room;
      cur := concat(cur, copy(s, i, take));
      conBuf^[conCnt] := cur;
      i := i + take;
      PaintRow(conCnt);
   end;
end;

{ Advance to a new empty bottom line in the console }
procedure ConNewLn;
var j: integer; wasBot: boolean;
begin
   CaretSet(false);
   wasBot := (topLine >= MaxTopLn);
   if conCnt < conCap then begin
      conCnt := conCnt + 1;
      conBuf^[conCnt] := '';
      if wasBot then begin
         if conCnt > topLine + visRows - 1 then
            ScrollToLn(MaxTopLn)
         else
            SyncElev;
      end;
   end
   else begin
      for j := 1 to conCap - 1 do
         conBuf^[j] := conBuf^[j + 1];
      conBuf^[conCap] := '';
      if wasBot and (myWindow <> nil) then begin
         InstallPort(myWindow);
         ClipTerm;
         ScrollRect(termRect, 0, -lineHt, scrollRgn);
         PaintRow(conCap);
         SyncElev;
      end;
   end;
end;

procedure ConWrLn(s: Str255);
begin
   if length(s) > 0 then ConWr(s);
   ConNewLn;
end;

procedure ConWrNum(n: longint);
var s: Str255;
begin
   IntToStr(n, s);
   ConWr(s);
end;

procedure ShowPrompt;
begin
   if length(conBuf^[conCnt]) > 0 then ConNewLn;
   cmdInput := '';
   conBuf^[conCnt] := '> ';
   atPrompt := true;
   SnapBottom;
   PaintRow(conCnt);
   blinkTime := Time;
end;

procedure SetPromptTxt(s: Str255);
begin
   CaretSet(false);
   if length(s) > ConCols - 2 then
      cmdInput := copy(s, 1, ConCols - 2)
   else
      cmdInput := s;
   conBuf^[conCnt] := concat('> ', cmdInput);
   SnapBottom;
   PaintRow(conCnt);
end;

procedure ConClear;
begin
   CaretSet(false);
   conCnt := 1;
   topLine := 1;
   conBuf^[1] := '';
   atPrompt := false;
   if myWindow <> nil then begin
      PaintTerm;
      SyncElev;
   end;
   ShowPrompt;
end;

{ Check the WM event queue during blocking serial waits / transfers.
  Apple-period sets aborted := true.  Window updates, activations, and Filer
  events are dispatched live so the desktop never freezes. }
function UserAbort: boolean;
var e: EventRecord; saveFldr: WindowPtr;
begin
   if not aborted then begin
      while EventAvail and (not aborted) and (not toldToDie) do begin
         GetEvent(e);
         if e.what = keyDown then begin
            if e.appleKey and (e.ascii = '.') then begin
               aborted := true;
               ConNewLn;
               pcol := 0;
               ConWrLn('*** Apple-period: stopping ***');
            end;
         end
         else if (e.what = folderUpdate) or (e.what = folderActivate) or
                 (e.what = folderDeactivate) or (e.what = filerEvent) then begin
            saveFldr := theFolder;
            inModal := true;
            ProcessEvent(e);
            inModal := false;
            if (myWindow = nil) or toldToDie then
               aborted := true
            else if saveFldr <> nil then
               InstallPort(saveFldr);
         end;
      end;
   end;
   UserAbort := aborted;
end;

{ Wait for one key in the dialog band while keeping updates & Filer healthy }
function ReadDlgCh(var ch: char): boolean;
var e: EventRecord; saveFldr: WindowPtr; gotKey: boolean;
begin
   ReadDlgCh := false;
   gotKey := false;
   repeat
      while not PeekEvent(e) do begin
         if toldToDie or (myWindow = nil) then exit(ReadDlgCh);
         LetOthersRun;
      end;
      if (e.what = buttonUp) or (e.what = folderUpdate) or
         (e.what = folderMoved) or (e.what = filerEvent) then begin
         GetEvent(e);
         saveFldr := theFolder;
         inModal := true;
         ProcessEvent(e);
         inModal := false;
         if saveFldr <> nil then InstallPort(saveFldr);
         if toldToDie or (myWindow = nil) then exit(ReadDlgCh);
      end
      else if e.what <> keyDown then
         exit(ReadDlgCh)
      else if e.appleKey then begin
         GetEvent(e);
         if e.ascii = '.' then begin
            ch := KbdClear;
            ReadDlgCh := false;
         end;
         exit(ReadDlgCh);
      end
      else begin
         GetEvent(e);
         ch := e.ascii;
         gotKey := true;
      end;
   until gotKey;
   ReadDlgCh := true;
end;

function AskDlgStr(prompt: Str255; defVal: Str255; allowEmpty: boolean;
                   var ans: Str255): boolean;
label 99;
var saveFldr: WindowPtr; ch: char; s1: string[1]; sw: integer;
    t: Rect; useDef: boolean; fullP: Str255;
begin
   AskDlgStr := false;
   useDef := (length(defVal) > 0);
   fullP := prompt;
   if useDef then fullP := concat(prompt, ' [', defVal, '] ');
   DrawDlgBox(fullP);
   saveFldr := theFolder;
   InstallPort(dialogFolder);
   if useDef then GetPen(promptPt);
   s1 := ' ';
   ans := '';
   repeat
      if not ReadDlgCh(ch) then goto 99;
      InstallPort(dialogFolder);
      TextFont(sysText);
      TextFace([]);
      if ch = KbdBS then begin
         if length(ans) > 0 then begin
            ch := ans[length(ans)];
            delete(ans, length(ans), 1);
            sw := CharWidth(ch);
            Move(-sw, 0);
            SetRect(t, dialogFolder^.pnLoc.h, dialogFolder^.portRect.top,
                    dialogFolder^.pnLoc.h + sw, dialogFolder^.portRect.bottom);
            EraseRect(t);
         end;
      end
      else if ch = KbdClear then
         goto 99
      else if ch >= ' ' then begin
         if (length(ans) = 0) and useDef then begin
            DrawDlgBox(prompt);
            InstallPort(dialogFolder);
            TextFont(sysText);
            TextFace([]);
            useDef := false;
         end;
         if length(ans) < 200 then begin
            DrawChar(ch);
            s1[1] := ch;
            ans := concat(ans, s1);
         end;
      end;
   until (ch = KbdCR) or (ch = KbdEnter);
   if (length(ans) = 0) and useDef then ans := defVal;
   TrimStr(ans);
   AskDlgStr := allowEmpty or (length(ans) > 0);
99:
   HideDlgBox;
   if saveFldr <> nil then InstallPort(saveFldr);
end;

function AskYes(question: Str255): boolean;
var ch: char; ok: boolean;
begin
   ConWr(concat(question, ' (Y/N) '));
   DrawDlgBox(concat(question, ' (Y or N) '));
   ch := 'N';
   repeat
      ok := ReadDlgCh(ch);
   until (not ok) or (ch in ['Y', 'y', 'N', 'n', KbdClear]);
   HideDlgBox;
   if ok and (ch in ['Y', 'y']) then begin
      ConWrLn('Y');
      AskYes := true;
   end
   else begin
      ConWrLn('N');
      AskYes := false;
   end;
end;

{ Create a Lisa .TEXT file and prefill its 1024-byte header page in obuf[1..1024] }
function OpenText(name: pathname; var fref: integer): boolean;
var j: integer; hdr: TxtHdrPtr; ns: Str255;
begin
   OpenText := false;
   ns := name;
   ecode := 0;
   BeginIO;
   MAKE_FILE(ecode, name, 0);
   EndIO;
   if ecode > 0 then begin
      ConWr(concat('cannot create ', ns, ': '));
      ConWrNum(ecode);
      ConNewLn;
      exit(OpenText);
   end;
   ecode := 0;
   BeginIO;
   OPEN(ecode, name, fref, [dwrite]);
   EndIO;
   if ecode > 0 then begin
      ConWr(concat('cannot open local file ', ns, ': '));
      ConWrNum(ecode);
      ConNewLn;
      exit(OpenText);
   end;
   { Build a clean Workshop/LOS .TEXT header page in obuf[1..1024] }
   for j := 1 to TextPgSz do big.obuf[j] := 0;
   hdr := pointer(ord4(@big.obuf[1]) + 512);
   hdr^.Version := TxtHdrVer;
   hdr^.FontId.Fam := p12Tile;
   hdr^.FontId.SetEFace := [];
   hdr^.FontId.Dev := devScreen;
   hdr^.Dummy1 := 0;
   hdr^.FontName := '';
   hdr^.TabsIncr := 8;
   hdr^.LMargin := 0;
   hdr^.RMargin := 0;
   hdr^.ParaMargin := 0;
   hdr^.CmdChar := '^';
   hdr^.DocPosn := 0;
   hdr^.ElevPosn := 0;
   hdr^.Select.LeftPl.Ln := 0;
   hdr^.Select.LeftPl.Ch := 1;
   hdr^.Select.RightPl := hdr^.Select.LeftPl;
   hdr^.Dirty := false;
   hdr^.TimeWritten := 0;
   hdr^.NumMarkers := 0;
   opos := TextPgSz;
   ooff := 0;
   opErr := 0;
   OpenText := true;
end;

procedure PageStart;
begin
   pgLines := 0;
   pageAll := false;
end;

function PageMore: boolean;
var ch: char; ok: boolean; lim: integer;
begin
   PageMore := true;
   if UserAbort then begin
      PageMore := false;
      exit(PageMore);
   end;
   if pageAll then exit(PageMore);
   lim := visRows - 2;
   if lim < 6 then lim := 6;
   if pgLines >= lim then begin
      pgLines := 0;
      DrawDlgBox('SPACE / RETURN = next page,  A = show all,  CLEAR / Q = stop');
      repeat
         ok := ReadDlgCh(ch);
      until (not ok) or (ch in [' ', KbdCR, KbdEnter, 'A', 'a', 'Q', 'q', KbdClear]);
      HideDlgBox;
      if (not ok) or (ch in ['Q', 'q', KbdClear]) then begin
         ConWrLn('(listing stopped)');
         PageMore := false;
         exit(PageMore);
      end;
      if ch in ['A', 'a'] then pageAll := true;
   end;
   pgLines := pgLines + 1;
end;


procedure ShowVolFree(volPfx: e_name);
var fsi: fs_info; p: pathname; ec: integer; vs: Str255;
begin
   p := volPfx;
   vs := volPfx;
   ec := 0;
   BeginIO;
   LOOKUP(ec, p, fsi);
   EndIO;
   if ec = 0 then begin
      ConWr(concat('Volume ', vs, '  ('));
      ConWrNum(fsi.freecount);
      ConWr(' free blocks, ~');

end;

procedure DoListVol(arg: Str255);
var err, j: integer;
    catPath: pathname;
    volPfx, prefix: e_name;
    uarg, outl, tn, attr, dm, dcr, szStr, pszStr: Str255;
    q: Q_Info;
    n: longint;
    more: boolean;
begin
   uarg := arg;
   TrimStr(uarg);
   volPfx := curVol;
   { Allow '!list -UPPER-' or '!list -UPPER-FOO' to inspect another volume }
   if (length(uarg) >= 2) and (uarg[1] = '-') then begin
      j := 2;
      while (j <= length(uarg)) and (uarg[j] <> '-') do j := j + 1;
      if (j <= length(uarg)) and (uarg[j] = '-') then begin
         if j <= MaxEName then volPfx := copy(uarg, 1, j);
         uarg := copy(uarg, j + 1, length(uarg) - j);
      end;
   end;
   if length(uarg) > 0 then
      if uarg[length(uarg)] = '=' then
         uarg := copy(uarg, 1, length(uarg) - 1);
   if length(uarg) > MaxEName then
      uarg := copy(uarg, 1, MaxEName);
   prefix := uarg;
   catPath := volPfx;
   err := 0;
   BeginIO;
   RESET_CATALOG(err, catPath);
   EndIO;
   if err <> 0 then begin
      outl := volPfx;
      ConWr(concat('!list: cannot read catalog of ', outl, ': '));
      ConWrNum(err);
      ConNewLn;
      exit(DoListVol);
   end;
   PageStart;
   ShowVolFree(volPfx);
   outl := volPfx;
   if PageMore then
      ConWrLn(concat('Contents of ', outl, uarg, '='));
   if PageMore then
      ConWrLn('Filename                            Size Psize   Last-Mod-Date  Creation-Date Attr')
   else exit(DoListVol);
   if PageMore then
      ConWrLn('--------                            ---- -----   -------------  ------------- ----')
   else exit(DoListVol);

   n := 0;
   more := true;
   repeat
      err := 0;
      BeginIO;
      LOOKUP_NEXT_ENTRY(err, prefix, q);
      EndIO;
      if err = 0 then begin
         n := n + 1;
         outl := q.name;
         PadRight(outl, MaxEName);
         attr := '';
         if q.etype <> fileentry then begin
            EntryKind(q.etype, tn);
            attr := tn;
         end;
         if q.protected then
            if attr = '' then attr := 'protected'
            else attr := concat(attr, ' protected');
         if attr <> '' then attr := concat(' ', attr);
         DateStr(q.DTM, dm);
         DateStr(q.DTC, dcr);
         PadLeftNum(q.size, 8, szStr);
         PadLeftNum((q.psize + 511) div 512, 6, pszStr);
         more := PageMore;
         if more then
            ConWrLn(concat(outl, szStr, pszStr, '  ', dm, ' ', dcr, attr));
      end;
   until (err <> 0) or (not more);
   if not more then exit(DoListVol);
   if (err <> 848) and (err <> 0) then
      if PageMore then begin
         ConWr('!list: catalog error: ');
         ConWrNum(err);
         ConNewLn;
      end;
   if PageMore then begin
      ConWrNum(n);
      ConWrLn(' entries');
   end;
end;

procedure DoDelLocal(name: Str255);
var p: pathname; ps: Str255; j: integer;
begin
   TrimStr(name);
   if name = '' then begin
      ConWrLn('usage: !delete <local>   (or !del)');
      exit(DoDelLocal);
   end;
   for j := 1 to length(name) do
      if (name[j] = '{') or (name[j] = '}') then begin
         ConWrLn(concat('refusing to delete ', name, ': Office System desktop/tool file'));
         exit(DoDelLocal);
      end;
   FullPath(name, p);
   ps := p;
   if not AskYes(concat('Delete ', ps, ' on the Lisa?')) then begin
      ConWrLn('not deleted');
      exit(DoDelLocal);
   end;
   ecode := 0;
   BeginIO;
   KILL_OBJECT(ecode, p);
   EndIO;
   if ecode > 0 then begin
      ConWr(concat('cannot delete ', ps, ': error '));
      ConWrNum(ecode);
      ConNewLn;
   end
   else begin
      ConWrLn(concat('deleted ', ps));
      SetStatus(concat('Deleted local file ', ps));
   end;
end;

procedure ProbeVol(pfx: e_name);
var fsi: fs_info; ec, dup, k: integer; path: pathname; s, ns: Str255;
begin
   if volCnt >= MaxVols then exit(ProbeVol);
   for k := 1 to volCnt do
      if volList[k] = pfx then exit(ProbeVol);
   path := pfx;
   ec := 0;
   BeginIO;
   LOOKUP(ec, path, fsi);
   EndIO;
   if ec <> 0 then exit(ProbeVol);
   if not fsi.mounted then exit(ProbeVol);
   dup := 0;
   for k := 1 to volCnt do
      if volFree[k] = fsi.freecount then dup := k;
   volCnt := volCnt + 1;
   volList[volCnt] := pfx;
   volFree[volCnt] := fsi.freecount;
   IntToStr(volCnt, ns);
   s := pfx;
   ConWr(concat('  ', ns, '  ', s, '  '));
   ConWrNum(fsi.freecount);
   ConWr(' free blocks (~');
   ConWrNum(fsi.freecount div 2);
   ConWr(' KB)');
   if pfx = myVol then ConWr('  [tool volume]');
   if dup > 0 then begin
      IntToStr(dup, ns);
      ConWr(concat('  (same free count as #', ns, ')'));
   end;
   ConNewLn;
end;

procedure ScanVols;
var pfx: e_name;
begin
   ConWrLn('Mounted Lisa volumes:');
   volCnt := 0;
   ProbeVol(myVol);
   pfx := '-#12-';   ProbeVol(pfx);
   pfx := '-#2#1-';  ProbeVol(pfx);
   pfx := '-#2#2-';  ProbeVol(pfx);
   pfx := '-#3#1-';  ProbeVol(pfx);
   pfx := '-#3#2-';  ProbeVol(pfx);
   pfx := '-#1#1-';  ProbeVol(pfx);
   pfx := '-#1#2-';  ProbeVol(pfx);
   pfx := '-LOWER-'; ProbeVol(pfx);
   pfx := '-UPPER-'; ProbeVol(pfx);
end;

procedure ApplyVol(choice: Str255);
var k, j: integer; allDig: boolean; p: pathname; fsi: fs_info; ec: integer; vs: Str255;
begin
   TrimStr(choice);
   if choice = '' then exit(ApplyVol);
   allDig := true;
   for j := 1 to length(choice) do
      if (choice[j] < '0') or (choice[j] > '9') then allDig := false;
   if allDig then begin
      k := StrToInt(choice);
      if (k >= 1) and (k <= volCnt) then begin
         curVol := volList[k];
         vs := curVol;
         ConWrLn(concat('Working volume is now ', vs));
         DrawHdrBar;
         SetStatus(concat('Working volume: ', vs));
      end
      else
         ConWrLn('not one of the listed volume numbers');
      exit(ApplyVol);
   end;
   if choice[1] <> '-' then choice := concat('-', choice);
   if choice[length(choice)] <> '-' then choice := concat(choice, '-');
   UpperStr(choice);
   if length(choice) > MaxEName then begin
      ConWrLn('volume prefix too long');
      exit(ApplyVol);
   end;
   p := choice;
   ec := 0;
   BeginIO;
   LOOKUP(ec, p, fsi);
   EndIO;
   if (ec <> 0) or (not fsi.mounted) then begin
      ConWr(concat('volume ', choice, ' is not mounted (error '));
      ConWrNum(ec);
      ConWrLn(')');
      exit(ApplyVol);
   end;
   curVol := choice;
   ConWrLn(concat('Working volume is now ', choice));
   DrawHdrBar;
   SetStatus(concat('Working volume: ', choice));
end;

procedure DoVolCmd(arg: Str255);
var vs: Str255;
begin
   TrimStr(arg);
   if arg = '' then begin
      ScanVols;
      vs := curVol;
      ConWrLn(concat('Current working volume: ', vs, '  (change with: !volume <number | -vol->)'));
   end
   else begin
      if volCnt = 0 then ScanVols;
      ApplyVol(arg);
   end;
end;

function SetupPort: boolean;
var e: integer; mvol, mpw, mdev: e_name; opn: pathname; ds: Str255;
begin
   if portOpen then begin
      SetupPort := true;
      exit(SetupPort);
   end;
   SetupPort := false;
   mvol := '';
   mpw := '';
   mdev := devName;
   e := 0;
   BeginIO;
   MOUNT(e, mvol, mpw, mdev);
   EndIO;
   if (e > 0) and (e <> 1052) then begin
      ds := devName;
      ConWr(concat('cannot mount ', ds, ' (error '));
      ConWrNum(e);
      ConWrLn(')');
      if e = 1051 then begin
         ConWrLn('Serial B is not configured on this boot volume:');
         ConWrLn('Open Preferences on the desktop -> Connect Devices ->');
         ConWrLn('set Serial B Connector to Serial Cable, then choose Connect.');
      end;
      SetStatus('Serial B not configured -- check Preferences > Connect Devices');
      exit(SetupPort);
   end;

   dcPath := concat('-', devName);
   e := DCpTry(1, 0, 0, 0);
   if e <> 0 then begin
      ConWr('RS232B device_control error: ');
      ConWrNum(e);
      ConNewLn;
      SetStatus('RS232B configuration failed');
      exit(SetupPort);
   end;
   DCp(5, curBaud, 0, 0);
   DCp(6, 0, 0, 0);
   DCp(7, 0, 0, 0);
   DCp(9, -1, -1, 32767);
   DCp(9, 1024, -1, 32767);
   DCp(10, 0, 0, 0);
   DCp(14, 0, 0, 0);
   curTmo := 0;
   DCp(25, 0, 0, 0);
   DCp(11, 0, 0, 0);
   DCp(12, 0, 0, 0);
   DCp(17, 0, 0, 0);
   DCp(19, 0, 0, 0);

   opn := concat(dcPath, '-HI');
   e := 0;
   BeginIO;
   OPEN(e, opn, portRef, [dread, dwrite]);
   EndIO;
   if e > 0 then begin
      ConWr('cannot open -RS232B-HI (error ');
      ConWrNum(e);
      ConWrLn(')');
      if (e = 946) or (e = 1051) then begin
         ConWrLn('Serial B is not set to Serial Cable in Preferences:');
         ConWrLn('Open Preferences -> Connect Devices -> Serial B = Serial Cable.');
      end;
      SetStatus('Cannot open Serial B -- set Serial B to Serial Cable in Preferences');
      exit(SetupPort);
   end;
   portOpen := true;
   SetupPort := true;
end;

         connected := false;
      end;
      e := 0;
      BeginIO;
      CLOSE_OBJECT(e, portRef);
      EndIO;
      portOpen := false;
      curTmo := -1;
   end;
end;

procedure DoBench(arg: Str255);
const NB = 10;
var j, k, fref, want, c2: integer;
    nbytes, got: longint;
    fname, bname: pathname;
    line, fs: Str255;
begin
   fname := '';
   if arg <> '' then begin
      FullPath(arg, fname);
      fs := fname;
      ecode := 0;
      BeginIO;
      LOOKUP(ecode, fname, finfo);
      EndIO;
      if ecode > 0 then begin
         ConWr(concat('bench: cannot open local file ', fs, ': '));
         ConWrNum(ecode);
         ConNewLn;
         exit(DoBench);
      end;
   end;
   if not ReadyLink then exit(DoBench);
   ConWrLn('bench: timing client operations (server is clock)');
   for j := 1 to ChunkSz do
      buf[j] := ToByte(((j mod 256) * 37) mod 256);
   nbytes := NB;
   nbytes := nbytes * ChunkSz;
   benchOh := 0;
   BenchStart;
   BenchMark('measurement overhead', 0);
   benchOh := benchMs;

   BenchStart;
   for k := 1 to 20 do SendRaw('NEXT');
   BenchMark('send 20 short lines (WRITE_DATA)', 120);

   BenchStart;
   SendRaw('ECHO 20');
   for k := 1 to 20 do
      if ReplyLine(line, 5, false, 9) <> WaitOK then ;
   BenchMark('receive 20 short lines (read-ahead)', 180);

   ProgStart(nbytes);
   holdAck := false;
   ackDue := false;
   BenchStart;
   SendRaw('FRAMES 10');
   xf.seq := 0;
   k := 0;
   repeat
      if k < NB then want := ChunkSz else want := 0;
      xf.ok := RecvChunk(xf.seq, xf.len, @buf[1], want);
      k := k + 1;
   until (not xf.ok) or (xf.len = 0);
   BenchMark('receive 10 chunks + QuickDraw progress', nbytes);
   ProgEnd;

   BenchStart;
   c2 := $FFFF;
   for k := 1 to NB do CRC16A(@buf[1], ChunkSz, c2, ord4(@crcTab));
   BenchMark('CRC-16 in 68000 assembly (CRC16A)', nbytes);

   BenchStart;
   xf.bsum := 0;
   for k := 1 to NB do SUM16A(@buf[1], ChunkSz, xf.bsum);
   BenchMark('byte sum in 68000 assembly (SUM16A)', nbytes);

   FullPath('LosSerialFtpClientBench.BIN', bname);
   ecode := 0;
   BeginIO;
   KILL_OBJECT(ecode, bname);
   EndIO;
   BenchStart;
   ecode := 0;
   BeginIO;
   MAKE_FILE(ecode, bname, 0);
   if ecode <= 0 then OPEN(ecode, bname, fref, [dwrite]);
   EndIO;
   if ecode <= 0 then begin
      got := 0;
      for k := 1 to (NB * ChunkSz) div WriteBlkSz do begin
         BeginIO;
         WRITE_DATA(ecode, fref, ord4(@big.obuf[1]), WriteBlkSz, actual, sequential, 0);
         EndIO;
         got := got + WriteBlkSz;
      end;
      BeginIO;
      CLOSE_OBJECT(ecode, fref);
      EndIO;
      BenchMark('write binary file (4 KB per WRITE_DATA)', got);
   end;
   BeginIO;
   KILL_OBJECT(ecode, bname);
   EndIO;

   if arg <> '' then begin
      fs := fname;
      BenchStart;
      if LocalSum(fname, not IsTextName(fs), false) then ;
      BenchMark('-c local check (read + decode + CRC + sum)', xf.total);
   end;
   ConWrLn('bench done');
   SetStatus('Benchmark finished');
end;

procedure RunCmdLine{(line: Str255)};
begin
   aborted := false;
   ParseCmd(line, cmdWord, argA, argB);
   if (cmdWord = 'REPEAT') or (cmdWord = 'R') then begin
      if lastCmd = '' then begin
         ConWrLn('no command to repeat');
         exit(RunCmdLine);
      end;
      ConWrLn(concat('repeating: ', lastCmd));
      line := lastCmd;
      ParseCmd(line, cmdWord, argA, argB);
   end;
   if cmdWord = '' then exit(RunCmdLine);
   cmdKnown := true;
   if (cmdWord = 'Q') or (cmdWord = 'EXIT') then cmdWord := 'QUIT';

   if cmdWord = 'LIST' then begin
      if ReadyLink then DoList(argA);
   end
   else if cmdWord = 'GET' then
      GetCmd(argA, argB)
   else if cmdWord = 'PUT' then
      PutCmd(argA, argB)
   else if (cmdWord = '!LIST') or (cmdWord = 'LLIST') then
      DoListVol(argA)
   else if (cmdWord = 'DELETE') or (cmdWord = 'DEL') then
      DoDelRemote(argA)
   else if (cmdWord = '!DELETE') or (cmdWord = '!DEL') then
      DoDelLocal(argA)
   else if (cmdWord = '!VOLUME') or (cmdWord = '!VOL') then
      DoVolCmd(argA)
   else if cmdWord = 'BAUD' then
      DoBaud(argA)
   else if cmdWord = 'HELLO' then begin
      if DoHello then ;
   end
   else if cmdWord = 'TIMEOUT-FACTOR' then
      DoTFactor(argA)
   else if cmdWord = 'DEBUG' then
      DoDebug(argA)
   else if cmdWord = 'BENCH' then
      DoBench(argA)
   else if (cmdWord = 'CLEAR') or (cmdWord = 'CLS') then
      ConClear
   else if (cmdWord = 'HELP') or (cmdWord = 'H') or (cmdWord = '?') then
      ShowHelp
   else if cmdWord = 'ABOUT' then
      ShowAbout
   else if cmdWord = 'QUIT' then begin
      PutAwayDoc;
      exit(RunCmdLine);
   end
   else begin
      ConWrLn(concat('unknown command: ', cmdWord, '  (type help for commands)'));
      cmdKnown := false;
   end;
   if cmdKnown and (cmdWord <> 'CLEAR') and (cmdWord <> 'CLS') then
      lastCmd := line;
end;

procedure DoScroll(startPt: Point; hitSB: THSB; hitIcon: TIcon);
var thumbPos, pgStep: integer;
begin
   BeginUpdate(myWindow);
   if not EmptyRgn(myWindow^.visRgn) then begin
      DrawChrome;
      PaintTerm;
   end;
   EndUpdate(myWindow);
   if hitSB <> rightSBox then exit(DoScroll);
   pgStep := visRows - 1;
   if pgStep < 1 then pgStep := 1;
   case hitIcon of
      IconArwA, IconArwB: begin
         ClipRect(myWindow^.portRect);
         PaintArw(hitSB, hitIcon, true);
         ClipTerm;
         repeat
            if hitIcon = IconArwA then ScrollToLn(topLine - 1)
            else ScrollToLn(topLine + 1);
         until not StillDown;
         ClipRect(myWindow^.portRect);
         PaintArw(hitSB, hitIcon, false);
         ClipTerm;
      end;
      IconThumb: begin
         ClipRect(myWindow^.portRect);
         DragThumb(hitSB, startPt, thumbPos);
         ClipTerm;
         ScrollToLn(CalcTopFromElev(thumbPos));
      end;
      IconPagA, IconPagB: begin
         ClipRect(myWindow^.portRect);
         PaintArw(hitSB, hitIcon, true);
         ClipTerm;
         repeat
            if hitIcon = IconPagA then ScrollToLn(topLine - pgStep)
            else ScrollToLn(topLine + pgStep);
         until not StillDown;
         ClipRect(myWindow^.portRect);
         PaintArw(hitSB, hitIcon, false);
         ClipTerm;
      end;
      otherwise ;
   end;
end;

procedure TrackCursor;
var savePort: GrafPtr; mousePt: Point; frame: Rect;
begin
   if (not isActive) or (myWindow = nil) then exit(TrackCursor);
   GetPort(savePort);
   SetPort(deskPort);
   GetMouse(mousePt);
   SetPort(savePort);
   InstallPort(myWindow);
   frame := myWindow^.portRect;
   with myWindow^.portBits.bounds do OffsetRect(frame, -left, -top);
   with deskPort^.portBits.bounds do OffsetRect(frame, left, top);
   if PtInRect(mousePt, frame) then begin
      SubPt(deskPort^.portBits.bounds.topLeft, mousePt);
      AddPt(myWindow^.portBits.bounds.topLeft, mousePt);
      if PtInRect(mousePt, termRect) then SetStdCursor(icrsXIbeam)
      else SetStdCursor(icrsInactive);
   end
   else
      SetStdCursor(icrsInactive);
end;

procedure HandleKey(var e: EventRecord);
var theMenu, theItem: integer; ch: char; subLine: Str255;
begin
   if (not isActive) or (myWindow = nil) then exit(HandleKey);
   InstallPort(myWindow);
   ch := e.ascii;
   if e.appleKey then begin
      RefreshMenus;
      MenuKey(ch, theMenu, theItem);
      if theMenu <> 0 then begin
         HiLiteMenu(theMenu);
         MenuCommand(theMenu, theItem);
      end;
      exit(HandleKey);
   end;
   if not atPrompt then exit(HandleKey);

   if IcrsCurrent = icrsXIbeam then ObscureCursor;
   SnapBottom;
   if (ch = KbdCR) or (ch = KbdEnter) then begin
      CaretSet(false);
      subLine := cmdInput;
      atPrompt := false;
      ConNewLn;
      TrimStr(subLine);
      if length(subLine) > 0 then
         RunCmdLine(subLine);
      if (myWindow <> nil) and (not toldToDie) then begin
         RefreshMenus;
         ShowPrompt;
      end;
   end
   else if ch = KbdBS then begin
      if length(cmdInput) > 0 then begin
         delete(cmdInput, length(cmdInput), 1);
         SetPromptTxt(cmdInput);
      end;
   end
   else if ch = KbdClear then
      SetPromptTxt('')
   else if ch = KbdUp then begin
      if length(lastCmd) > 0 then SetPromptTxt(lastCmd);
   end
   else if ch = KbdDown then
      SetPromptTxt('')
   else if (ch >= ' ') and (length(cmdInput) < ConCols - 3) then begin
      AddChar(cmdInput, ch);
      SetPromptTxt(cmdInput);
   end;
   CaretSet(true);
   blinkTime := Time + CaretOnTime;
end;

procedure HandleFilerOp(var e: EventRecord; var wantsOpen: boolean);
var fx: FilerExt; err: integer;
begin
   wantsOpen := false;
   GetAddParms(err, e, fx);
   case fx.theFlrOp of
      fcNone:
         wantsOpen := true;
      fcResume:
         TellFiler(err, docClosd, internalError, e.who);
      fcClose, fcSuspend, fcShred: begin
         ShutPort;
         if e.who = myWindow then ForgetWindow;
         TellFiler(err, docClosd, allOK, e.who);
      end;
      fcCopy, fcPut:
         TellFiler(err, docNotXfered, internalError, e.who);
      fcDfClose:
         TellFiler(err, dfClosed, allOK, e.who);
      fcTerminate:
         toldToDie := true;
      otherwise ;
   end;
end;

procedure ProcessEvent{(var e: EventRecord)};
var theMenu, theItem: integer;
    hitSB: THSB;
    hitIcon: TIcon;
    wantsOpen: boolean;
begin
   if (myWindow <> nil) and (e.who = myWindow) then InstallPort(myWindow);

   case e.what of
      buttonDown:
         if not inModal then begin
            if e.who = menuFolder then begin
               RefreshMenus;
               SetStdCursor(icrsInactive);
               MenuSelect(e.where, theMenu, theItem);
               if (theMenu <> 0) and (theItem <> 0) then
                  MenuCommand(theMenu, theItem)
               else
                  HiLiteMenu(0);
            end
            else if (e.who = myWindow) and isActive then begin
               if FGrowHit(e.where) then
                  DoGrow(e.where)
               else if FSbHit(docSBList, e.where, hitSB, hitIcon) then
                  DoScroll(e.where, hitSB, hitIcon)
               else if PtInRect(e.where, termRect) then begin
                  SnapBottom;
                  CaretSet(true);
                  blinkTime := Time + CaretOnTime;
               end;
            end;
         end;

      buttonUp:
         ;

      keyDown:
         if not inModal then HandleKey(e);

      folderActivate: begin
         TakeControl(e, false, false);
         if (myWindow <> nil) and (e.who = myWindow) then begin
            InstallPort(myWindow);
            isActive := true;
            PutUpMenus;
            DrawChrome;
            caretVis := false;
            blinkTime := Time;
         end;
      end;

      folderDeactivate: begin
         if (myWindow <> nil) and (e.who = myWindow) and isActive then begin
            InstallPort(myWindow);
            CaretSet(false);
            UnDrawChrome;
            isActive := false;
            SetStdCursor(icrsInactive);
         end;
         GiveControl(e);
      end;

      folderUpdate:
         if (myWindow <> nil) and (e.who = myWindow) then begin
            InstallPort(myWindow);
            CaretSet(false);
            BeginUpdate(myWindow);
            RecompGeom;
            ClipRect(myWindow^.portRect);
            EraseRect(myWindow^.portRect);
            if isActive then DrawChrome else UnDrawChrome;
            PaintTerm;
            EndUpdate(myWindow);
            ClipTerm;
         end
         else begin
            BeginUpdate(e.who);
            EndUpdate(e.who);
         end;

      filerEvent: begin
         HandleFilerOp(e, wantsOpen);
         if wantsOpen then InitNewDoc(e.who);
      end;

      otherwise ;
   end;
end;

procedure InitDlgFont;
var fInfo: FontInfo; errNum: integer;
begin
   deskStyle.Fam := sysText;
   deskStyle.SetEFace := [];
   deskStyle.Dev := devScreen;
   if FMFontMetrics(deskStyle, fInfo, errNum) then
      with fInfo do begin
         boxHeight := ascent + descent + 12;
         boxVOff := (boxHeight + ascent - descent) div 2;
         boxHOff := widMax;
      end
   else begin
      boxHeight := 24;
      boxVOff := 16;
      boxHOff := 8;
   end;
end;

procedure Initialize;
var err, junk, cutAt, openBr: integer;
    pcInfo: procinfoRec;
    excName: t_ex_name;
    segAddr: longint;
    braceStr: string[1];
begin
   initing := true;
   ioNest := 0;

   { (1) SYS_TERMINATE handler FIRST (calls ImDying) }
   excName := 'SYS_TERMINATE';
   Declare_Excep_Hdl(err, excName, @LastWishes);
   if err <> 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      Halt;
   end;

   { (2) Initialize all state }
   toldToDie := false;
   myWindow := nil;
   theFolder := nil;
   isActive := false;
   needAutoCn := false;
   sbMade := false;
   statOK := false;
   statMsg := 'Ready  --  Type a command or use the menus (Apple-. cancels)';
   barActive := false;
   barPct := 0;
   elevPos := 0;
   origTitle := '';
   shownTitle := '';
   atPrompt := false;
   cmdInput := '';
   caretVis := false;
   blinkTime := 0;
   dlgOpen := false;
   inModal := false;
   fontChoice := miFont15;
   termFam := p15Tile;
   lineHt := 12;
   ascentPx := 9;
   charW := 7;

   devName := 'RS232B';
   portOpen := false;
   curBaud := DefBaud;
   tfactor := 1;
   curTmo := -1;
   rxqn := 0;
   rxqp := 1;
   holdAck := false;
   ackDue := false;
   linkDead := false;
   eofDone := false;
   pcol := 0;
   pdone := 0;
   ptotal := 0;
   debugOn := false;
   dbSync := false;
   lastCmd := '';
   logRx := true;
   connected := false;
   aborted := false;
   dirty := false;
   volCnt := 0;

   { (3) Private heap for menus, alerts, scroll bars, regions }
   InitHeap(@myZone[0], @myZone[MaxHeapW], @OutOfHeapSpace);

   { (4) Build CRC-16 lookup table and verify 68000 assembly routines }
   if not CrcSelfTest then begin
      TellFiler(err, initFailed, internalError, nil);
      Halt;
   end;

   { (5) Discover our installed pathname prefix & tool volume }
   BeginIO;
   Info_Process(err, My_ID, pcInfo);
   EndIO;
   cutAt := pos('}', pcInfo.progpathname);
   if (err <> 0) or (cutAt = 0) or (cutAt > 40) then begin
      TellFiler(err, initFailed, internalError, nil);
      Halt;
   end;
   myPrefix := copy(pcInfo.progpathname, 1, cutAt);

   braceStr := ' ';
   braceStr[1] := chr(123);   { open brace }
   openBr := pos(braceStr, pcInfo.progpathname);
   if (openBr > 1) and (openBr <= MaxEName + 1) then
      myVol := copy(pcInfo.progpathname, 1, openBr - 1)
   else
      myVol := '-#12-';
   curVol := myVol;

   { (6) Load menus and alerts from <myPrefix>PHRASE }
   InitMP(theHeap, myPrefix, @AlertFailed, AlertVer, @myMenus[mFile], allAlerts, err);
   if err > 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      Halt;
   end;

   { (7) Window Manager cursors & scroll bars }
   InitWmlCrs(err);
   if err <> 0 then begin
      TellFiler(err, initFailed, internalError, nil);
      Halt;
   end;
   InitWmlSb;
   SetPort(deskPort);
   InitDlgFont;
   scrollRgn := NewRgn;

   { (8) Allocate scrollback console buffer in a private data segment       }
   {     (falls back to a smaller heap block if disk space is tight)        }
   conSegPath := concat(myPrefix, 'ftp console');
   hasConSeg := false;
   BeginIO;
   Kill_DataSeg(junk, conSegPath);
   Make_DataSeg(err, conSegPath, ConSegBytes, ConSegBytes, conSegRef,
                segAddr, ConLDSN, ds_private);
   EndIO;
   if err <= 0 then begin
      hasConSeg := true;
      conBuf := pointer(segAddr);
      conCap := ConMaxLn;
   end
   else begin
      conBuf := pointer(ord4(NewPtr(ConFallLn * sizeof(ConLine))));
      if conBuf = nil then begin
         TellFiler(err, initFailed, noDiskSpace, nil);
         Halt;
      end;
      conCap := ConFallLn;
   end;
   conCnt := 1;
   topLine := 1;
   conBuf^[1] := '';

   initing := false;
end;

{$S }

begin
   InitGraf(@thePort);
   OpenWM;
   Initialize;

   repeat
      { Phase A: inactive -- block in GetEvent }
      while not (ImActive or toldToDie) do begin
         GetEvent(theEvent);
         ProcessEvent(theEvent);
      end;

      { Phase B: active -- poll events, blink caret, track cursor, yield }
      while ImActive and (not toldToDie) do begin
         if EventAvail then begin
            GetEvent(theEvent);
            ProcessEvent(theEvent);
         end
         else begin
            if isActive and (myWindow <> nil) then begin
               if needAutoCn then begin
                  needAutoCn := false;
                  CaretSet(false);
                  atPrompt := false;
                  if not DoHello then
                     ConWrLn('(server not yet responding; local commands work, choose Connect to retry)');
                  RefreshMenus;
                  ShowPrompt;
               end;
               TrackCursor;
               if atPrompt and (Time >= blinkTime) then begin
                  CaretSet(not caretVis);
                  if caretVis then blinkTime := Time + CaretOnTime
                  else blinkTime := Time + CaretOffTime;
               end;
            end;
            LetOthersRun;
         end;
      end;
   until toldToDie;

   ShutPort;
   if hasConSeg then begin
      BeginIO;
      Close_DataSeg(schedErr, conSegRef);
      Kill_DataSeg(schedErr, conSegPath);
      EndIO;
   end;
end.
